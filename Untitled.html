options(repos = c(CRAN = “https://cloud.r-project.org”)) install.packages(“ggplot2”) install.packages(“gridExtra”)

#Parameters and IC
N <- 1e6
N_half <- N / 2
nsteps <- 100
Delta <- 1
S <- 1
tau <- 1
set.seed(0329)
x <- rep(0, N)
dir <- c(rep(-1, N_half), rep(1, N_half))   # -1 = left, +1 = right
#simulating the random walks
for (step in 1:nsteps) {
  new_dir <- sample(c(-1, 1), size = N, replace = TRUE)
  shift <- ifelse(dir == -1 & new_dir == 1,  S,
           ifelse(dir ==  1 & new_dir == -1, -S, 0))
  x <- x + new_dir * Delta + shift
  dir <- new_dir
}
#n's
binwidth <- 1
xmin <- min(x) - 2
xmax <- max(x) + 2
breaks <- seq(xmin - 0.5, xmax + 0.5, by = binwidth)
centers <- head(breaks, -1) + binwidth / 2
x_r <- x[dir == 1]
x_l <- x[dir == -1]
hist_r <- hist(x_r, breaks = breaks, plot = FALSE)
hist_l <- hist(x_l, breaks = breaks, plot = FALSE)
n_r_num <- hist_r$counts
n_l_num <- hist_l$counts
n_num <- n_r_num + n_l_num
delta_num <- n_r_num - n_l_num
#eq26 results
t_final <- nsteps * tau
D0 <- Delta^2 / (2 * tau)
L0 <- sqrt(D0 * t_final)
n_theory <- N / (sqrt(4 * pi) * L0) * exp(-(centers^2) / (4 * L0^2))
delta_theory <- centers * (Delta + S / 2) / (2 * L0^2) * n_theory
plot_df <- data.frame(
  x = centers,
  n_num = n_num,
  delta_num = delta_num,
  n_theory = n_theory,
  delta_theory = delta_theory
)

p1 <- ggplot(plot_df, aes(x = x)) +
  geom_line(aes(y = n_num), linewidth = 0.35, color = "blue") +
  geom_point(aes(y = n_num), size = 0.5, color = "blue") +
  geom_line(aes(y = n_theory), linewidth = 0.5, color = "red") +
  labs(
    title = "total",
    x = "x",
    y = "n(x, 100τ)"
  ) +
  theme_minimal(base_size = 16)

p2 <- ggplot(plot_df, aes(x = x)) +
  geom_line(aes(y = delta_num), linewidth = 0.35, color = "blue") +
  geom_point(aes(y = delta_num), size = 0.5, color = "blue") +
  geom_line(aes(y = delta_theory), linewidth = 0.5, color = "red") +
  labs(
    title = "imbalance",
    x = "x",
    y = expression(delta*n(x,100*tau) == n[r] - n[l])
  ) +
  theme_minimal(base_size = 16)
grid.arrange(p1, p2, ncol = 2)
